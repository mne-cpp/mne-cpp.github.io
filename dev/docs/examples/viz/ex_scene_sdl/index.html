Run the trace view in an SDL3 window instead of Qt

Trace view in an SDL3 window, without Qt: mne_scene's view and software renderer in another toolkit.

ex_scene_sdlIntermediateVisualize dataReal-time pipelinesEEG

Libraries: Visualization (SDL), Visualization, FIFF I/O

What it demonstrates​

The same TraceView, TraceBuffer and TraceFrameBuilder that drive the raw browser of mne_disp run in an SDL3 window. Every frame is executed by the software backend of the GPU frame description and shown as an SDL texture; SDL mouse, wheel and key events are translated to scene input events, so wheel zoom, drag scrolling with inertia, the ruler (right button) and the crosshair (c) work as in the Qt host. A synthetic 32-channel recording streams in at 1 kHz and the view follows its end until the user scrolls.

With --panel the window shows a tool panel instead (projectors, the default, or compensator) through FormView, which draws and edits any SettingsForm without Qt and without a view written for the panel. The average and spectrum panels show their figures, synthetic averages at their sensor positions or synthetic spectra, through FigureView, which zooms, pans or scrolls and passes the pointer to any SceneFigure.

The channels panel shows the item lists of the channel selection: its groups, their channels and the channels picked by hand.

Usage: ex_scene_sdl [--panel [projectors|compensator|average|spectrum|channels]] [--frames N]; with --frames the example renders N frames and exits (SDL_VIDEODRIVER=dummy runs it without a display).

Build and run​

cmake --build build/developer-dynamic --target ex_scene_sdl
./out/Release/examples/ex_scene_sdl

using namespace mne::viz;

constexpr int kChannels = 32;
constexpr double kSfreq = 1000.0;

/** The next block of the synthetic recording: alpha-like oscillations, a slow drift and an occasional spike. */
Eigen::MatrixXd synthetic_block(int first, int samples)
{
    Eigen::MatrixXd block(kChannels, samples);
    for (int ch = 0; ch < kChannels; ++ch) {
        const double frequency = 6.0 + 0.4 * ch;
        for (int s = 0; s < samples; ++s) {
            const double t = (first + s) / kSfreq;
            double value = std::sin(2.0 * std::numbers::pi * frequency * t + ch) + 0.3 * std::sin(2.0 * std::numbers::pi * 0.3 * t);
            if ((first + s) % 2000 < 30 && ch % 4 == 0) {
                value += 2.5;
            }
            block(ch, s) = value * 20e-6;
        }
    }
    return block;
}

//=============================================================================================================
/**
 * Shows the averages of a synthetic evoked response at the positions of a 4 x 4 grid of EEG sensors through
 * FigureView: the wheel zooms, the right button pans, a double click fits and pressing a tile enlarges its name.
 *
 * @param[in] host      The window.
 * @param[in] metrics   Metrics of the fonts the host draws with.
 *
 * @return The number of frames drawn.
 */
int run_average_layout(sdl::SdlHost& host, const TextMetrics& metrics)
{
    constexpr int kSensors = 16;
    constexpr int kSamples = 300;
    constexpr int kPreStimulus = 100;
    //! [average_layout_panel_usage]
    // An N100-like deflection that grows towards the back of the grid, and a later positive wave
    auto info = std::make_shared<mne::FiffInfo>();
    Eigen::MatrixXd auditory = Eigen::MatrixXd::Zero(kSensors, kSamples);
    std::vector<AverageLayoutFigure::Channel> selection;
    for (int i = 0; i < kSensors; ++i) {
        const int row = i / 4;
        const int column = i % 4;
        mne::FiffChInfo channel;
        channel.ch_name = "EEG " + std::to_string(i + 1);
        channel.kind = FIFFV_EEG_CH;
        channel.unit = FIFF_UNIT_V;
        info->chs.push_back(channel);
        info->ch_names.push_back(channel.ch_name);
        for (int s = kPreStimulus; s < kSamples; ++s) {
            const double t = (s - kPreStimulus) / kSfreq;
            const double n100 = -std::exp(-std::pow((t - 0.1) / 0.02, 2));
            const double p300 = 0.6 * std::exp(-std::pow((t - 0.3) / 0.05, 2));
            auditory(i, s) = (n100 * (1.0 + row) + p300) * 5e-6;
        }
        AverageLayoutFigure::Channel tile;
        tile.name = channel.ch_name;
        tile.number = i;
        tile.position = {1.6 * column, -1.2 * row};
        selection.push_back(std::move(tile));
    }
    info->nchan = kSensors;

    AverageLayoutPanel panel;
    panel.set_info(info);
    panel.set_selection(std::move(selection));
    panel.set_scale_map({{FIFFV_EEG_CH, 1e-5f}});
    panel.set_average_colors({{"auditory", Color{31, 111, 235}}});
    EvokedAverages evoked;
    evoked.averages = {{"auditory", auditory.data(), kSamples, kSensors}};
    evoked.first_sample = -kPreStimulus;
    evoked.bad.assign(kSensors, false);
    evoked.bad[5] = true;
    panel.set_evoked(std::move(evoked));

    // Any panel's figure, without a view written for it
    FigureView view(*panel.figure());
    view.set_text_metrics(&metrics);
    return host.run(view);
    //! [average_layout_panel_usage]
}

//=============================================================================================================
/**
 * Shows the spectra of eight synthetic channels with an alpha peak through FigureView: the wheel scrolls the rows,
 * the pointer reads out a bin and a double click freezes the spectra.
 *
 * @param[in] host      The window.
 * @param[in] metrics   Metrics of the fonts the host draws with.
 *
 * @return The number of frames drawn.
 */
int run_spectrum(sdl::SdlHost& host, const TextMetrics& metrics)
{
    constexpr int kSpectrumChannels = 8;
    constexpr int kBins = 256;
    //! [spectrum_panel_usage]
    auto info = std::make_shared<mne::FiffInfo>();
    info->sfreq = kSfreq;
    Eigen::MatrixXd spectra(kSpectrumChannels, kBins);
    for (int i = 0; i < kSpectrumChannels; ++i) {
        mne::FiffChInfo channel;
        channel.ch_name = "EEG " + std::to_string(i + 1);
        info->chs.push_back(channel);
        info->ch_names.push_back(channel.ch_name);
        // 1/f background in dB with an alpha peak that grows towards the back of the head
        for (int bin = 0; bin < kBins; ++bin) {
            const double frequency = bin * (kSfreq / 2.0) / (kBins - 1);
            spectra(i, bin) = -10.0 * std::log10(frequency + 1.0) + (1.0 + i) * std::exp(-std::pow((frequency - 10.0) / 1.5, 2));
        }
    }
    info->nchan = kSpectrumChannels;

    SpectrumPanel panel;
    panel.init(info, SpectrumTable::Scale::Linear);
    panel.add_data(spectra);
    panel.set_boundaries(1.0f, 60.0f);

    FigureView view(*panel.figure());
    view.set_text_metrics(&metrics);
    return host.run(view);
    //! [spectrum_panel_usage]
}

//=============================================================================================================
/**
 * Shows the channel selection through FormView: the layout and group files, the selection groups, the channels of
 * the chosen group and the picked channels as lists that a click, the arrow keys and Delete edit.
 *
 * @param[in] host      The window.
 * @param[in] metrics   Metrics of the fonts the host draws with.
 *
 * @return The number of frames drawn.
 */
int run_channel_selection(sdl::SdlHost& host, const TextMetrics& metrics)
{
    //! [channel_selection_panel_usage]
    // The Vectorview layout and the mne_browse_raw groups of the installed resources
    ChannelSelectionPanel panel;
    ChannelSelectionPanel::Callbacks callbacks;
    callbacks.channels_shown = [](const std::vector<std::string>& channels) {
        std::printf("Showing %zu channels\n", channels.size());
    };
    panel.set_callbacks(std::move(callbacks));
    panel.set_mapped_channels({"MEG 0111", "MEG 0112", "MEG 0113", "MEG 0121"});
    panel.select_channels({"MEG 0112"});

    // Any panel's lists, without a view written for them
    FormView view(panel);
    view.set_text_metrics(&metrics);
    return host.run(view);
    //! [channel_selection_panel_usage]
}

//=============================================================================================================
/**
 * Shows a tool panel through FormView, the toolkit-free view of any SettingsForm, without a view of its own.
 *
 * @param[in] host      The window.
 * @param[in] which     "projectors", "compensator", "average", "spectrum" or "channels".
 * @param[in] metrics   Metrics of the fonts the host draws with.
 *
 * @return The number of frames drawn.
 */
int run_panel(sdl::SdlHost& host, const std::string& which, const TextMetrics& metrics)
{
    if (which == "average") {
        return run_average_layout(host, metrics);
    }
    if (which == "spectrum") {
        return run_spectrum(host, metrics);
    }
    if (which == "channels") {
        return run_channel_selection(host, metrics);
    }
    //! [form_view_usage]
    std::unique_ptr<Panel> panel;
    if (which == "compensator") {
        auto compensator = std::make_unique<CompensatorPanel>();
        std::vector<mne::FiffCtfComp> comps(3);
        for (int i = 0; i < 3; ++i) {
            comps[static_cast<std::size_t>(i)].kind = i + 1;
        }
        compensator->set_compensators(comps);
        compensator->set_selection_callback([](int to) {
            std::printf("Compensation grade %d\n", to);
        });
        panel = std::move(compensator);
    } else {
        auto projectors = std::make_unique<ProjectorsPanel>();
        std::vector<mne::FiffProj> projs(4);
        const char* descriptions[] = {"Average EEG reference", "PCA-v1 (planar)", "PCA-v2 (axial)", "ECG-planar-999--0.200-0.400-PCA-01"};
        for (std::size_t i = 0; i < projs.size(); ++i) {
            projs[i].desc = descriptions[i];
            projs[i].active = i != 2;
        }
        projectors->set_projectors(projs);
        projectors->set_selection_callback([](const std::vector<mne::FiffProj>& selected) {
            std::printf("Projectors:");
            for (const mne::FiffProj& proj : selected) {
                std::printf(" %d", proj.active ? 1 : 0);
            }
            std::printf("\n");
        });
        panel = std::move(projectors);
    }
    FormView view(*panel);
    view.set_text_metrics(&metrics);
    return host.run(view);
    //! [form_view_usage]
}

} // namespace

//=============================================================================================================
// MAIN
//=============================================================================================================

int main(int argc, char* argv[])
{
    sdl::SdlHost::Options options;
    options.title = "MNE-CPP trace view (SDL3, no Qt)";
    std::string panel;
    for (int i = 1; i < argc; ++i) {
        if (std::strcmp(argv[i], "--panel") == 0) {
            const bool named = i + 1 < argc && std::strncmp(argv[i + 1], "--", 2) != 0;
            panel = named ? argv[++i] : "projectors";
            options.title = "MNE-CPP " + panel + " panel (SDL3, no Qt)";
            options.width = 420;
            options.height = 360;
        } else if (std::strcmp(argv[i], "--frames") == 0 && i + 1 < argc) {
            const std::string_view value(argv[i + 1]);
            if (std::from_chars(value.data(), value.data() + value.size(), options.frame_limit).ec != std::errc()) {
                std::fprintf(stderr, "--frames expects a number, got %s\n", argv[i + 1]);
                return 1;
            }
        }
    }

    //! [sdl_host_usage]
    sdl::SdlHost host(options);
    if (!host.is_open()) {
        std::fprintf(stderr, "Cannot open the window: %s\n", host.error().c_str());
        return 1;
    }
    // The layouts measure the text with the fonts the host draws it with
    const FontTextMetrics metrics(options.fonts);
    //! [sdl_host_usage]

    if (!panel.empty()) {
        const int frames = run_panel(host, panel, metrics);
        std::printf("Rendered %d frames of the %s panel with the SDL3 backend of mne_viz\n", frames, panel.c_str());
        return 0;
    }

    // The channels and the first seconds of the recording
    TraceBuffer buffer;
    std::vector<TraceBuffer::ChannelSpec> specs;
    for (int ch = 0; ch < kChannels; ++ch) {
        TraceBuffer::ChannelSpec spec;
        spec.name = "EEG " + std::to_string(ch + 1);
        spec.type_label = "EEG";
        spec.color = Color{170, 55, 10};
        spec.default_scale = 40e-6f;
        spec.bad = ch == 5;
        specs.push_back(std::move(spec));
    }
    buffer.set_channels(std::move(specs));
    buffer.set_sample_rate(static_cast<float>(kSfreq));
    buffer.set_capacity(60 * static_cast<int>(kSfreq));
    int recorded = 4 * static_cast<int>(kSfreq);
    buffer.set_data(synthetic_block(0, recorded), 0);

    //! [trace_view_client_usage]
    TraceViewClient client;
    client.style().background = Color{250, 250, 250};
    TraceView& view = client.view();
    view.set_sfreq(kSfreq);
    view.set_first_file_sample(0);
    view.set_visible_channel_count(16);
    view.set_samples_per_pixel(5.0);
    view.set_scalebars_visible(true);
    client.set_source(&buffer);
    client.set_text_metrics(&metrics);
    //! [trace_view_client_usage]

    // The view follows the end of the recording until the user scrolls
    bool followEnd = true;
    bool following = false; // the scroll below is the example's, not the user's
    TraceView::Effects effects;
    effects.scroll_changed = [&](double /*sample*/) {
        followEnd = followEnd && following;
    };
    client.set_effects(std::move(effects));

    host.set_key_handler([&](const KeyEvent& key) {
        if (key.is_character(U'c')) {
            view.set_crosshair_enabled(!view.crosshair_enabled());
            return true;
        }
        if (key.key == Key::End) {
            followEnd = true;
            return true;
        }
        return false;
    });

    const int frames = host.run(client, [&](std::int64_t now) {
        // 1 kHz streaming: append what arrived since the last frame
        const int target = static_cast<int>(4 * kSfreq + static_cast<double>(now));
        if (target > recorded) {
            const int samples = std::min(target - recorded, 200);
            buffer.append_data(synthetic_block(recorded, samples));
            recorded += samples;
            client.data_changed();
        }
        if (followEnd) {
            const double visible = view.view_width() * view.samples_per_pixel();
            following = true;
            view.set_scroll_sample(std::max(0.0, recorded - visible));
            following = false;
        }
    });

    std::printf("Rendered %d frames with the SDL3 backend of mne_viz (SDL %d.%d.%d)\n", frames, SDL_MAJOR_VERSION, SDL_MINOR_VERSION, SDL_MICRO_VERSION);
    return 0;
}
